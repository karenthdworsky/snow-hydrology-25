1. Snow Metamorphism and Densification Model

The densification model tracks old and new snow layers separately: - Old snow density approaches ρ_max via exponential decay - New snow density uses the temperature-dependent equation: ρ = 50 + 1.7 × (T + 15)^(3/2)

rho_max     <- 350   # kg/m³ — maximum density (from SNOTEL observations)
k           <- 0.01  # densification rate (per hour)
density_old <- 250   # kg/m³ — initial old snow density
depth_old   <- 0
depth_new   <- 0
density_new <- 0

depth_total   <- numeric(nrow(asos_data))
density_total <- numeric(nrow(asos_data))

new_snow_density <- function(temp_C) {
  pmax(50, pmin(400, 50 + 1.7 * (temp_C + 15)^(3/2)))
}

for (i in seq_len(nrow(asos_data))) {
  temp_C     <- asos_data$temp_C[i]
  snowfall_m <- asos_data$snowfall_m[i]

  density_old <- rho_max - (rho_max - density_old) * exp(-k)
  depth_old   <- depth_old * density_old / (density_old + k * depth_old)

  if (!is.na(temp_C) && !is.na(snowfall_m) && snowfall_m > 0) {
    density_new <- new_snow_density(temp_C)
    depth_new   <- depth_new + snowfall_m * 100  # convert m → cm
  }

  total_depth <- depth_old + depth_new
  depth_total[i] <- total_depth
  if (total_depth > 0) {
    density_total[i] <- (depth_old * density_old + depth_new * density_new) / total_depth
  } else {
    density_total[i] <- density_old
  }

  if (!is.na(snowfall_m) && snowfall_m > 0) {
    depth_old   <- depth_total[i]
    density_old <- density_total[i]
    depth_new   <- 0
    density_new <- 0
  }
}

asos_data$modeled_depth_cm   <- depth_total
asos_data$modeled_density    <- density_total

asos_data <- asos_data %>%
  mutate(
    snowfall_cm        = snowfall_m * 100,
    cumulative_snowfall_cm = cumsum(snowfall_cm)
  )

4. Hedstrom-Pomeroy Fresh Snow Density

hedstrom_pomeroy <- function(temp_C) {
  pmax(50, pmin(200, 67.92 + 51.25 * exp(temp_C / 2.59)))
}

5. Observed vs. Modeled Bulk Density (SNOTEL) — Sturm et al. 2010

snotel_data <- read.table("data/SNOWTEL.txt", header = TRUE, sep = "\t")
colnames(snotel_data) <- trimws(colnames(snotel_data))
snotel_data$Date  <- as.Date(snotel_data$Date, format = "%m/%d/%Y")
snotel_data       <- snotel_data[order(snotel_data$Date), ]

snotel_data <- snotel_data %>%
  mutate(
    depth_m     = SNWD_in * 0.0254,
    swe_m       = WTEQ_in * 0.0254,
    obs_density = ifelse(depth_m > 0, swe_m / depth_m, NA),
    doy         = row_number()
  )

sturm_density <- function(depth_cm, doy) {
  rho_max <- 0.5975
  k1      <- 0.0012
  k2      <- 0.0038
  rho_0   <- 0.024
  rho_max * (1 - exp(-k1 * depth_cm - k2 * doy)) + rho_0
}

snotel_data <- snotel_data %>%
  mutate(sturm_density_gcc = sturm_density(depth_m * 100, doy))

ggplot(snotel_data, aes(x = Date)) +
  geom_point(aes(y = obs_density), color = "#2166ac", size = 2, alpha = 0.7) +
  geom_line(aes(y = sturm_density_gcc), color = "#d73027", linewidth = 1) +
  labs(title = "Observed vs. Modeled Bulk Snow Density (WY2024)",
       subtitle = "Red Mountain Pass SNOTEL Site 713 | Sturm et al. 2010 (alpine)",
       caption = "Observed (blue) | Modeled (red)",
       x = "Date", y = "Bulk Density (g/cm³)") +
  theme_minimal() +
  theme(plot.title = element_text(hjust = 0.5, face = "bold"))
