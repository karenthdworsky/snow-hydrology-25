2. Undercatch Correction

RAIN_CATCH_EFFICIENCY <- 0.95

estimate_snow_catch_efficiency <- function(wind_speed_ms) {
  catch_eff <- ifelse(is.na(wind_speed_ms), 0.70,
                      1.00 - (wind_speed_ms * 0.08))
  pmax(0.40, pmin(0.95, catch_eff))
}

precip_corrected <- precip_wy24 %>%
  filter(!is.na(p01m), p01m > 0) %>%
  mutate(
    precip_type = case_when(
      is.na(temp_c)             ~ "Unknown",
      temp_c <= 0               ~ "Snow",

      temp_c > 3                ~ "Rain",
      TRUE ~ "Unknown"
    ),
    snow_catch_eff   = estimate_snow_catch_efficiency(wind_ms),
    catch_efficiency = case_when(
      precip_type == "Rain"  ~ RAIN_CATCH_EFFICIENCY,
      precip_type == "Snow"  ~ snow_catch_eff,
      precip_type == "Mixed" ~ (RAIN_CATCH_EFFICIENCY + snow_catch_eff) / 2,
      TRUE ~ 0.80
    ),
    p01m_corrected    = p01m / catch_efficiency,
    correction_amount = p01m_corrected - p01m
  )

precip_complete <- precip_wy24 %>%
  left_join(
    precip_corrected %>% select(datetime, precip_type, catch_efficiency,

  mutate(
    p01m           = ifelse(is.na(p01m), 0, p01m),

    cumulative_uncorrected = cumsum(p01m),
    cumulative_corrected   = cumsum(p01m_corrected)

3. Precipitation as Snow — Three Methods

TEMP_ALL_SNOW <- -2
TEMP_ALL_RAIN <-  4

calc_snow_probability <- function(temp_c) {
  temp_mid  <- (TEMP_ALL_SNOW + TEMP_ALL_RAIN) / 2
  temp_range <- TEMP_ALL_RAIN - TEMP_ALL_SNOW
  k          <- 10 / temp_range
  prob_snow  <- 1 / (1 + exp(k * (temp_c - temp_mid)))
  prob_snow  <- ifelse(temp_c <= TEMP_ALL_SNOW, 1.0, prob_snow)
  prob_snow  <- ifelse(temp_c >= TEMP_ALL_RAIN, 0.0, prob_snow)
  ifelse(is.na(temp_c), NA_real_, prob_snow)
}

precip_snow_analysis <- precip_complete %>%
  mutate(
    dewpoint_c = as.numeric(ifelse(dwpc == "M" | dwpc == "" | is.na(dwpc), NA, dwpc)),
    method1_is_snow  = !is.na(temp_c) & temp_c <= 0,
    method1_snow_mm  = ifelse(method1_is_snow, p01m_corrected, 0),
    method1_rain_mm  = ifelse(!method1_is_snow, p01m_corrected, 0),
    method2_is_snow  = !is.na(dewpoint_c) & dewpoint_c <= 0,
    method2_snow_mm  = ifelse(method2_is_snow, p01m_corrected, 0),
    method2_rain_mm  = ifelse(!method2_is_snow, p01m_corrected, 0),
    snow_probability = calc_snow_probability(temp_c),
    method3_snow_mm  = ifelse(!is.na(snow_probability),
                              p01m_corrected * snow_probability, 0),
    method3_rain_mm  = ifelse(!is.na(snow_probability),
                              p01m_corrected * (1 - snow_probability), 0),
    cumulative_method1 = cumsum(method1_snow_mm),
    cumulative_method2 = cumsum(method2_snow_mm),
    cumulative_method3 = cumsum(method3_snow_mm)
  )

total_precip  <- max(precip_snow_analysis$cumulative_corrected, na.rm = TRUE)

assign_wy_month_factor <- function(date_val) {
  m <- month(date_val)
  factor(ifelse(m >= 10, m - 9, m + 3), levels = 1:12,
         labels = c("Oct","Nov","Dec","Jan","Feb","Mar",
                    "Apr","May","Jun","Jul","Aug","Sep"))
}

4. Wind During Precipitation Events

wind_analysis <- precip_snow_analysis %>%
  mutate(
    wind_ms = as.numeric(ifelse(is.na(wind_ms), NA, wind_ms)),
    precip_condition = case_when(
      is.na(temp_c) | is.na(p01m_corrected) ~ "Unknown",
      p01m_corrected <= 0                   ~ "No Precipitation",
      temp_c <= 0 & p01m_corrected > 0      ~ "Snow",
      temp_c > 0  & p01m_corrected > 0      ~ "Rain",
      TRUE ~ "Unknown"
    ),
    wy_month = assign_wy_month_factor(datetime)
  ) %>%
  filter(precip_condition != "Unknown")
