1. Melt Factor Calibration

A temperature-index melt equation was calibrated using SNOTEL SWE change data: M = MF × (T − T₀), where T > T₀

Non-linear least squares estimates MF (melt factor) and T₀ (base temperature).

2. Fit Temperature-Index Melt Equation

swe_filtered <- swe_data %>%
  filter(!is.na(change_SWE) & change_SWE != 0 &
         !is.na(temp_C) & temp_C <= 13 & temp_C > 0)

cat(sprintf("Observations for model fitting: %d\n", nrow(swe_filtered)))
Observations for model fitting: 165
melt_model <- tryCatch(
  nls(change_SWE ~ MF * (temp_C - T0),
      data    = swe_filtered,
      start   = list(MF = 2.0, T0 = 0),
      control = nls.control(maxiter = 1000, minFactor = 1e-6)),
  error = function(e) {
    message("NLS did not converge; using linear approximation: ", e$message)
    NULL
  }
)

if (!is.null(melt_model)) {
  mf_estimate <- coef(melt_model)[["MF"]]
  t0_estimate <- coef(melt_model)[["T0"]]
} else {
  lm_model    <- lm(change_SWE ~ temp_C, data = swe_filtered)
  mf_estimate <- coef(lm_model)[["temp_C"]]
  t0_estimate <- -coef(lm_model)[["(Intercept)"]] / mf_estimate
}

mf_estimate <- as.numeric(mf_estimate)
t0_estimate <- as.numeric(t0_estimate)

cat(sprintf("Melt factor (MF): %.4f mm/°C\n", mf_estimate))
Melt factor (MF): -0.0639 mm/°C
cat(sprintf("Base temperature (T0): %.2f °C\n", t0_estimate))
Base temperature (T0): 96.53 °C
cat(sprintf("Melt equation: M = %.4f × (T − %.2f)\n", mf_estimate, t0_estimate))
Melt equation: M = -0.0639 × (T − 96.53)
temp_seq <- seq(t0_estimate, 13, length.out = 100)
fit_line  <- data.frame(temp_C = temp_seq,
                         change_SWE = mf_estimate * (temp_seq - t0_estimate))

ggplot(swe_filtered, aes(x = temp_C, y = change_SWE)) +

3. Baseflow Separation — Lynne-Hollick Digital Filter

The Lynne-Hollick recursive digital filter separates streamflow into baseflow and quickflow components. Three passes improve baseflow isolation.

discharge_data <- read.csv("data/NWIS_discharge.csv", stringsAsFactors = FALSE)

discharge_data <- discharge_data %>%
  mutate(date     = mdy(time),
         flow_cfs = as.numeric(value)) %>%
  filter(!is.na(flow_cfs)) %>%
  arrange(date)

lynne_hollick_filter <- function(Q, a = 0.925) {
  bf    <- numeric(length(Q))
  bf[1] <- Q[1]
  for (i in 2:length(Q)) {
    bf[i] <- ((1 - a) / (1 + a)) * Q[i] + (a / (1 + a)) * bf[i - 1]
    bf[i] <- min(bf[i], Q[i])
  }
  bf
}

baseflow <- discharge_data$flow_cfs
for (i in 1:3) {
  baseflow <- lynne_hollick_filter(baseflow)
}

discharge_data$baseflow_cfs   <- baseflow
discharge_data$quickflow_cfs  <- discharge_data$flow_cfs - baseflow

total_volume    <- sum(discharge_data$flow_cfs)    * 86400 / 43559.9
baseflow_volume <- sum(discharge_data$baseflow_cfs) * 86400 / 43559.9

cat("=== BASEFLOW SEPARATION RESULTS ===\n")

5. Hourly Melt Estimation

Applying the calibrated melt factor to hourly ASOS temperature data:

asos_melt <- read.csv("data/asos_data.txt", stringsAsFactors = FALSE)

asos_melt <- asos_melt %>%
  mutate(
    valid  = ymd_hm(valid),
    tmpc   = as.numeric(tmpc),
    hourly_melt = ifelse(!is.na(tmpc) & tmpc > t0_estimate,
                         mf_estimate * (tmpc - t0_estimate), 0)
  ) %>%
  filter(!is.na(valid))
