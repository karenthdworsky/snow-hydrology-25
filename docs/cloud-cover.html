Cloud Cover Functions

cloud_cover_to_fraction <- function(sky_condition) {
  case_when(
    sky_condition == "SKC" | sky_condition == "CLR" ~ 0.0,
    sky_condition == "FEW" ~ 0.125,
    sky_condition == "SCT" ~ 0.375,
    sky_condition == "BKN" ~ 0.625,
    sky_condition == "OVC" ~ 1.0,
    TRUE ~ NA_real_
  )
}

categorize_cloud_cover <- function(sky_condition) {
  case_when(
    sky_condition == "SKC" | sky_condition == "CLR" ~ "Clear",
    sky_condition == "FEW" ~ "Few",
    sky_condition == "SCT" ~ "Scattered",
    sky_condition == "BKN" ~ "Broken",
    sky_condition == "OVC" ~ "Overcast",
    TRUE ~ "Missing"
  )
}

assign_wy_month <- function(datetime) {
  calendar_month <- month(datetime)
  wy_month_num   <- ifelse(calendar_month >= 10, calendar_month - 9, calendar_month + 3)
  factor(wy_month_num, levels = 1:12,
         labels = c("Oct","Nov","Dec","Jan","Feb","Mar",
                    "Apr","May","Jun","Jul","Aug","Sep"))
}

processed_data <- data_wy24 %>%
  mutate(
    datetime      = ymd_hm(valid),
    wy_month      = assign_wy_month(datetime),
    cloud_type_lowest   = categorize_cloud_cover(skyc1),
    cloud_fraction_lowest = cloud_cover_to_fraction(skyc1),
    skyc1_frac = cloud_cover_to_fraction(skyc1),
    skyc2_frac = cloud_cover_to_fraction(skyc2),
    skyc3_frac = cloud_cover_to_fraction(skyc3),
    all_decks_fraction = pmax(
      ifelse(is.na(skyc1_frac), 0, skyc1_frac),
      ifelse(is.na(skyc2_frac), 0, skyc2_frac),
      ifelse(is.na(skyc3_frac), 0, skyc3_frac),
      na.rm = TRUE
    ),
    cloud_type_all = case_when(
      all_decks_fraction == 0.0  ~ "Clear",
      all_decks_fraction <= 0.125 ~ "Few",
      all_decks_fraction <= 0.375 ~ "Scattered",
      all_decks_fraction < 1.0   ~ "Broken",
      all_decks_fraction == 1.0  ~ "Overcast",
      TRUE ~ "Missing"
    )
  ) %>%
  filter(!is.na(cloud_fraction_lowest))

hourly_data <- processed_data %>%
  group_by(wy_month, date = date(datetime), hour = hour(datetime)) %>%

4. Snow Crystal Morphology (WY2024)

classify_snow_crystal <- function(temp_c, supersaturation) {
  case_when(
    temp_c > 0 ~ "No Snow (Liquid)",
    temp_c >= -5 ~ case_when(
      supersaturation >= 0.15 ~ "Needles",
      supersaturation >= 0.05 ~ "Solid Plates",
      TRUE ~ "Thin Plates"),
    temp_c >= -10 ~ case_when(
      supersaturation >= 0.2  ~ "Needles",
      supersaturation >= 0.1  ~ "Hollow Columns",
      TRUE ~ "Solid Plates"),
    temp_c >= -15 ~ case_when(
      supersaturation >= 0.25 ~ "Stellars",
      supersaturation >= 0.15 ~ "Sectored Plates",
      supersaturation >= 0.1  ~ "Solid Plates",
      TRUE ~ "Thin Plates"),
    temp_c >= -22 ~ case_when(
      supersaturation >= 0.2 ~ "Stellars",
      supersaturation >= 0.1 ~ "Plates",
      TRUE ~ "Columns"),
    TRUE ~ "Columns"
  )
}

estimate_supersaturation <- function(temp_c, rh_percent) {
  case_when(
    is.na(rh_percent)   ~ NA_real_,
    rh_percent >= 95    ~ 0.25,
    rh_percent >= 85    ~ 0.15,
    rh_percent >= 70    ~ 0.08,
    TRUE                ~ 0.03
  )
}

snow_crystal_data <- data_wy24 %>%
  mutate(
    temp_c   = as.numeric(ifelse(tmpc == "M" | is.na(tmpc), NA, tmpc)),
    relh     = as.numeric(ifelse(relh == "M" | is.na(relh), NA, relh)),
    datetime = ymd_hm(valid),
    wy_month = assign_wy_month(datetime)
  ) %>%
  filter(!is.na(skyc1), skyc1 != "CLR", skyc1 != "SKC") %>%
  mutate(
    temp_at_cloud_base = temp_c,
    supersaturation  = estimate_supersaturation(temp_c, relh),
    crystal_type     = case_when(
      is.na(temp_at_cloud_base) | is.na(supersaturation) ~ "Unknown",
      TRUE ~ classify_snow_crystal(temp_at_cloud_base, supersaturation)
    ),
    crystal_category = case_when(
      crystal_type %in% c("Columns","Hollow Columns","Solid Prisms") ~ "Columns",
      crystal_type %in% c("Thin Plates","Solid Plates","Sectored Plates","Plates") ~ "Plates",
      crystal_type == "Needles"          ~ "Needles",
      crystal_type == "Stellars"         ~ "Stellars",
      crystal_type == "No Snow (Liquid)" ~ "No Snow",
      TRUE ~ "Unknown"
    )
  )

monthly_crystal_form <- snow_crystal_data %>%
  group_by(wy_month) %>%

    total_obs    = n(),

    avg_supersaturation  = mean(supersaturation, na.rm = TRUE),
    .groups = "drop"
  )
